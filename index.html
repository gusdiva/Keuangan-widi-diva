        const API_URL = "https://script.google.com/macros/s/AKfycbwAWStFkvXPcMsrj_0K8W5W3QlqXl9yX5TrK1ePh-ttziMTuylIg8ZTgjE_N8uVF6AK1w/exec";


        // Variabel global penyimpan data agar switch tab instan tanpa loading ulang

        let dataGlobalFinansial = null;

        let tabAktifSaatIni = "DIVA";


        // Atur tanggal default input form ke tanggal hari ini saat halaman dimuat

        document.addEventListener("DOMContentLoaded", () => {

        /**

         * Mengambil data terupdate dari Google Sheet lewat API

         */

        function ambilDataSistemDashboard() {

            const statusWaktu = document.getElementById("waktu-update");

            const panelError = document.getElementById("error-dashboard");

            statusWaktu.innerText = "Memperbarui data dari Google Sheets...";

            panelError.classList.add("hidden");


            fetch(API_URL, {

                method: "POST",

                mode: "cors",

                headers: { "Content-Type": "text/plain;charset=utf-8" },

                body: JSON.stringify({ action: "getDataDashboard" })

            })

            .then(res => res.json())

            .then(res => {

                if (res.error) {

                    panelError.innerText = res.error;

                    panelError.classList.remove("hidden");

                    statusWaktu.innerText = "Gagal memperbarui.";

                } else {

                    dataGlobalFinansial = res;

                    statusWaktu.innerText = "Terakhir diperbarui: " + res.waktuUpdate;

                    renderTabelDashboard(tabAktifSaatIni);

                }

            })

            .catch(err => {

                panelError.innerText = "Gagal terhubung ke server backend: " + err.toString();

                panelError.classList.remove("hidden");

                statusWaktu.innerText = "Koneksi terputus.";

            });

        }


        /**

         * Merender ulang tabel html berdasarkan tab nama yang dipilih

         */

        function renderTabelDashboard(nama) {

            if (!dataGlobalFinansial) return;

            

            document.getElementById("label-tabel-nama").innerText = nama;

            const barisData = nama === "DIVA" ? dataGlobalFinansial.diva : dataGlobalFinansial.widi;

            const containerTabel = document.getElementById("body-tabel-dashboard");

            

            let htmlElemen = "";

            

            // Loop data baris 1 sampai 4 (Pemasukan, Belanja, Cicilan, Tabungan)

            for (let r = 1; r <= 4; r++) {

                if (!barisData[r]) continue;

                htmlElemen += `<tr class="border-b border-slate-50">

                    <td class="py-2.5 text-slate-400">${barisData[r][0]}</td>

                    <td class="text-right py-2.5">${barisData[r][1] || 'Rp0'}</td>

                    <td class="text-right py-2.5">${barisData[r][2] || 'Rp0'}</td>

                    <td class="text-right py-2.5">${barisData[r][3] || 'Rp0'}</td>

                </tr>`;

            }

            

            // Tambahkan baris ke-5 sebagai Sisa Saldo dengan highlight khusus

            if (barisData[5]) {

                htmlElemen += `<tr class="bg-emerald-50/60 font-bold text-emerald-700">

                    <td class="p-2.5 rounded-l-xl">${barisData[5][0]}</td>

                    <td class="text-right p-2.5">${barisData[5][1] || 'Rp0'}</td>

                    <td class="text-right p-2.5">${barisData[5][2] || 'Rp0'}</td>

                    <td class="text-right p-2.5 rounded-r-xl">${barisData[5][3] || 'Rp0'}</td>

                </tr>`;

            }

            

            containerTabel.innerHTML = htmlElemen;

        }

        function switchTab(nama) {

            }

        /**

         * Mengirimkan Form Inputan Data Transaksi Baru ke Google Sheet

         */

        function handleSimpanTransaksi(event) {

            event.preventDefault();

            

            const btnSubmit = document.getElementById("btn-submit");

            btnSubmit.innerText = "Menyimpan ke Google Sheet...";

            btnSubmit.disabled = true;

            

            const paketData = {

                tanggal: document.getElementById("input-tanggal").value,

                nama: document.getElementById("input-nama").value,

                tipe: document.getElementById("input-tipe").value,

                kategori: document.getElementById("input-kategori").value,

                keterangan: document.getElementById("input-keterangan").value,

                jumlah: document.getElementById("input-jumlah").value

            };


            fetch(API_URL, {

                method: "POST",

                mode: "cors",

                headers: { "Content-Type": "text/plain;charset=utf-8" },

                body: JSON.stringify({ action: "tambahTransaksi", data: paketData })

            })

            .then(res => res.json())

            .then(res => {

                alert(res.pesan);

                if (res.sukses) {

                    document.getElementById("input-keterangan").value = "";

                    document.getElementById("input-file-ocr").value = "";

                    

                    // Tarik data dashboard paling baru agar angka tabel langsung update otomatis

                    ambilDataSistemDashboard();

                }

                btnSubmit.innerText = "Simpan Transaksi";

                btnSubmit.disabled = false;

            })

            .catch(err => {

                alert("Gagal menyimpan transaksi: " + err.toString());

                btnSubmit.innerText = "Simpan Transaksi";

                btnSubmit.disabled = false;

            });

        }


        /**

         * Menangani Upload Gambar & Proses OCR Teks Struk Belanja

         */

        function handleOcrScan(input) {

            const file = input.files[0];

            if (!file) return;


            const divStatus = document.getElementById("status-ocr");

            divStatus.className = "mt-2 text-xs font-semibold text-amber-600 flex items-center gap-1.5";

            divStatus.innerText = "⏳ Membaca teks struk belanja (OCR)...";

            divStatus.classList.remove("hidden");


            const reader = new FileReader();

            reader.readAsDataURL(file);

            reader.onload = function () {

                const hasilBase64Full = reader.result;

                const murniBase64 = hasilBase64Full.split(",")[1];

                

                fetch(API_URL, {

                    method: "POST",

                    mode: "cors",

                    headers: { "Content-Type": "text/plain;charset=utf-8" },

                    body: JSON.stringify({

                        action: "prosesOcrStruk",

                        data: { base64Data: murniBase64, mimeType: file.type }

                    })

                })

                .then(res => res.json())

                .then(res => {

                    if (res.sukses) {

                        divStatus.className = "mt-2 text-xs font-semibold text-emerald-600 flex items-center gap-1.5";

                        divStatus.innerText = "✅ Berhasil mendeteksi nominal angka!";

                        

                        // Masukkan nominal hasil deteksi otomatis ke kotak jumlah uang

                        if (res.jumlah > 0) {

                            document.getElementById("input-jumlah").value = res.jumlah;

                        } else {

                            divStatus.className = "mt-2 text-xs font-semibold text-amber-600 flex items-center gap-1.5";

                            divStatus.innerText = "⚠️ Gambar berhasil dibaca, tapi gagal menemukan nominal angka. Mohon input manual.";

                        }

                    } else {

                        divStatus.className = "mt-2 text-xs font-semibold text-rose-600 flex items-center gap-1.5";

                        divStatus.innerText = "❌ " + res.error;

                    }

                })

                .catch(err => {

                    divStatus.className = "mt-2 text-xs font-semibold text-rose-600 flex items-center gap-1.5";

                    divStatus.innerText = "❌ Gagal OCR: " + err.toString();

                });

            };

        }
